PlayerCombat --> EnemyHealth : dano melee

EnemyHealth ..> CombatEvents : RaiseEnemyKilled()

EnemyAI --> EnemyConfig

EnemyHealth --> EnemyConfig
EnemyHealth --> ItemData

EnemyHealth ..> MageTurretAI : desabilita ao morrer

BossController --> BossAttackDefinition
BossController --> EnemyHealth
BossController --> PlayerHealth
BossController ..> CombatTextManager

QuestManager --> QuestDefinition
QuestManager --> QuestManager_QuestState
QuestManager --> QuestManager_QuestOverride
QuestManager ..> InventoryManager : hook OnInventoryChanged
QuestManager ..> CombatEvents : subscreve OnEnemyKilled
QuestManager ..> GameCompleteUI : mostra fim de jogo

EnemySpawner ..> CombatEvents : subscreve OnEnemyKilled
EnemySpawner ..> QuestManager : IsQuestCompleted()
BossSpawnTrigger ..> BossController : Activate()

InventoryManager --> InventoryItem
InventoryItem --> ItemData

InventoryManager ..> CombatTextManager

PlayerHUD --> PlayerHealth
PlayerHUD --> PlayerProgression
EnemyHealthBar --> EnemyHealth
QuestHUD --> QuestManager
